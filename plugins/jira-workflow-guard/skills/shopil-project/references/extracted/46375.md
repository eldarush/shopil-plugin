Python / function tool; connection variables configured separately

import requests

from agents import function_tool

@function_tool

def push_tool(message: str) -> str:

"""Send the requested push notification."""

payload = {"user": pushover_user, "token": pushover_token,

"message": message}

response = requests.post(pushover_url, data=payload, timeout=15)

response.raise_for_status()

if response.json().get("status") != 1:

raise RuntimeError("Pushover did not accept the request")

return "Pushover accepted the notification."
